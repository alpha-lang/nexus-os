      {isHotel && stockDash && (
        <div>
          <div className="flex items-center justify-between mb-3">
            <div className="flex items-center gap-2">
              <div className="w-6 h-6 rounded-lg bg-teal-500 text-white flex items-center justify-center">
                <svg className="w-3.5 h-3.5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M20 7l-8-4-8 4m16 0l-8 4m8-4v10l-8 4m0-10L4 7m8 4v10M4 7v10l8 4"/>
                </svg>
              </div>
              <h2 className="font-black text-slate-900 text-sm uppercase tracking-wider">Stock</h2>
            </div>
            <a href="/dashboard/stock" className="text-[10px] font-black text-teal-600 hover:underline tracking-wider">
              VOIR TOUT
            </a>
          </div>

          {/* 3 mini KPI Stock */}
          <div className="grid grid-cols-3 gap-3 mb-3">
            <a href="/dashboard/stock" className="bg-slate-900 rounded-2xl p-4 text-white hover:shadow-lg transition">
              <p className="text-[9px] text-slate-400 uppercase font-black tracking-widest mb-1">Valorisation</p>
              <p className="text-xl font-black text-teal-400 tabular-nums">
                {Math.round(stockDash.summary?.totalValue || 0).toLocaleString('fr-FR')}
                <span className="text-xs text-slate-400 ml-1">Ar</span>
              </p>
              <p className="text-[9px] text-slate-500 mt-0.5">{stockDash.summary?.totalItems || 0} articles</p>
            </a>

            <a
              href="/dashboard/stock?alert=1"
              className={'rounded-2xl p-4 hover:shadow-lg transition ' + (
                (stockDash.alerts?.outOfStock?.length || 0) > 0
                  ? 'bg-red-500 text-white'
                  : (stockDash.alerts?.critical?.length || 0) > 0
                  ? 'bg-amber-500 text-white'
                  : 'bg-white border border-slate-200'
              )}
            >
              <p className={'text-[9px] uppercase font-black tracking-widest mb-1 ' + ((stockDash.alerts?.outOfStock?.length || 0) + (stockDash.alerts?.critical?.length || 0) > 0 ? 'text-white/80' : 'text-slate-500')}>Alertes</p>
              <p className={'text-xl font-black tabular-nums ' + ((stockDash.alerts?.outOfStock?.length || 0) + (stockDash.alerts?.critical?.length || 0) > 0 ? 'text-white' : 'text-slate-900')}>
                {(stockDash.alerts?.outOfStock?.length || 0) + (stockDash.alerts?.critical?.length || 0)}
              </p>
              <p className={'text-[9px] mt-0.5 ' + ((stockDash.alerts?.outOfStock?.length || 0) + (stockDash.alerts?.critical?.length || 0) > 0 ? 'text-white/80' : 'text-slate-400')}>
                {(stockDash.alerts?.outOfStock?.length || 0)} rupture{(stockDash.alerts?.outOfStock?.length || 0) > 1 ? 's' : ''} · {(stockDash.alerts?.critical?.length || 0)} critiques
              </p>
            </a>

            <a href="/dashboard/stock/movements" className="bg-white rounded-2xl p-4 border border-slate-200 hover:shadow-lg transition">
              <p className="text-[9px] text-slate-500 uppercase font-black tracking-widest mb-1">Mvts 30j</p>
              <p className="text-xl font-black text-slate-900 tabular-nums">{stockDash.summary?.movements30d || 0}</p>
              <p className="text-[9px] text-slate-400 mt-0.5">
                {(stockDash.summary?.incoming30d || 0) > 0 ? '+' + Math.round(stockDash.summary.incoming30d).toLocaleString('fr-FR') + ' Ar recus' : 'aucune entree'}
              </p>
            </a>
          </div>

          {/* Articles critiques + derniers mouvements */}
          <div className="grid grid-cols-1 lg:grid-cols-2 gap-3">
            {/* Critiques */}
            <div className="bg-white rounded-2xl border border-slate-200 overflow-hidden">
              <div className="px-4 py-3 border-b border-slate-100 flex items-center justify-between">
                <h3 className="font-bold text-slate-900 text-sm">Articles critiques</h3>
                {(stockDash.alerts?.critical?.length || 0) + (stockDash.alerts?.outOfStock?.length || 0) > 0 && (
                  <a href="/dashboard/stock/orders" className="text-[10px] font-black text-teal-600 hover:underline tracking-wider">
                    COMMANDER
                  </a>
                )}
              </div>
              <div className="divide-y divide-slate-100">
                {[...(stockDash.alerts?.outOfStock || []), ...(stockDash.alerts?.critical || [])].slice(0, 4).map((a: any) => (
                  <div key={a.id} className="px-4 py-2.5 flex items-center gap-3">
                    <div className={'w-2 h-2 rounded-full shrink-0 ' + ((a.currentStock || 0) <= 0 ? 'bg-red-500' : 'bg-amber-500')}></div>
                    <div className="flex-1 min-w-0">
                      <p className="font-bold text-slate-900 text-sm truncate">{a.name}</p>
                      <p className="text-[10px] text-slate-500">
                        {a.currentStock || 0} {a.unit || ''} · min {a.minStock || 0} {a.unit || ''}
                      </p>
                    </div>
                    <a
                      href={'/dashboard/stock/orders'}
                      className="text-[10px] font-black text-teal-600 hover:underline shrink-0"
                    >
                      Commander
                    </a>
                  </div>
                ))}
                {(stockDash.alerts?.critical?.length || 0) + (stockDash.alerts?.outOfStock?.length || 0) === 0 && (
                  <div className="px-4 py-8 text-center">
                    <p className="text-slate-400 text-xs">Tous les stocks sont OK</p>
                  </div>
                )}
              </div>
            </div>

            {/* Derniers mouvements */}
            <div className="bg-white rounded-2xl border border-slate-200 overflow-hidden">
              <div className="px-4 py-3 border-b border-slate-100 flex items-center justify-between">
                <h3 className="font-bold text-slate-900 text-sm">Derniers mouvements</h3>
                <a href="/dashboard/stock/movements" className="text-[10px] font-black text-teal-600 hover:underline tracking-wider">
                  VOIR TOUT
                </a>
              </div>
              <div className="divide-y divide-slate-100">
                {stockMvts.map((m: any) => {
                  const isOut = m.quantity < 0;
                  return (
                    <div key={m.id} className="px-4 py-2.5 flex items-center gap-3">
                      <div className={'w-0.5 self-stretch rounded-full shrink-0 ' + (isOut ? 'bg-red-500' : 'bg-emerald-500')}></div>
                      <div className="flex-1 min-w-0">
                        <p className="font-bold text-slate-900 text-sm truncate">{m.item?.name}</p>
                        <p className="text-[10px] text-slate-500 truncate">
                          {m.type === 'RECEPTION' ? 'Reception' : m.type === 'CONSUMPTION' ? 'Consommation' : m.type} · {new Date(m.createdAt).toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit' })}
                        </p>
                      </div>
                      <div className={'text-right font-black text-sm tabular-nums shrink-0 ' + (isOut ? 'text-red-600' : 'text-emerald-600')}>
                        {isOut ? '' : '+'}{m.quantity} <span className="text-[10px] text-slate-400 font-bold">{m.item?.unit}</span>
                      </div>
                    </div>
                  );
                })}
                {stockMvts.length === 0 && (
                  <div className="px-4 py-8 text-center">
                    <p className="text-slate-400 text-xs">Aucun mouvement recent</p>
                  </div>
                )}
              </div>
            </div>
          </div>
        </div>
      )}
